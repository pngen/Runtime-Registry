function(rr_add_example name)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE RuntimeRegistry::runtimeregistry)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
endfunction()
