function(rr_add_test name)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE RuntimeRegistry::runtimeregistry)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME ${name} COMMAND ${name})
endfunction()

rr_add_test(test_identity)
rr_add_test(test_registry)
rr_add_test(test_protocol)
rr_add_test(test_leases)
rr_add_test(test_query)
rr_add_test(test_adversarial)
rr_add_test(test_concurrency)
